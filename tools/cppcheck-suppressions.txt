# cppcheck suppressions for ntlibc, used by tools/lint.sh (`make lint`).
#
# Only categorical false positives belong here -- things cppcheck cannot
# model about a freestanding libc built with -nostdinc on top of ntdll.
# Findings that are merely inconvenient do not; they get reported and
# judged like any other.  Note that lint.sh runs cppcheck with
# --enable=warning,portability only, so the "prefer the _s variant" style
# advice aimed at *callers* of the C library never fires in the first
# place and needs no suppression here.
#
# Deliberately kept short and id-scoped rather than path-scoped: a
# `*:src/string/*.c` style blanket entry would silence real findings along
# with the noise, which defeats the point of running the tool.

# cppcheck cannot see the system headers we deliberately do not include
# (-nostdinc) and reports every one of them as missing.
missingInclude
missingIncludeSystem

# Inline assembly writes through pointer operands with a "memory" clobber.
# cppcheck's value-flow engine has no model for that and reports the
# results as uninitialised reads.
uninitvar:src/math/x87.h
uninitvar:arch/i386/src/*.c
uninitvar:arch/x86_64/src/*.c
uninitvar:src/setjmp/*

# Reported when a suppression above happens not to match on some arch or
# cppcheck version; not itself a finding about the code.
unmatchedSuppression

# We build with -nostdinc against our own <stdarg.h>, which defines
# va_start/va_arg/va_end as the __builtin_* forms the compilers actually
# implement.  cppcheck's va_list checker only recognises the names it
# expects from a system stdarg.h, so after preprocessing it sees a va_list
# that is never started and reports every use of it.  Applies to every
# variadic function in the tree, not to any particular one.
va_list_usedBeforeStarted

# printf.c reverses a digit buffer into a local (`rev`, `b2`) and emits
# exactly as many bytes as it wrote.  The digit count comes from a
# do/while, so it is >= 1, but cppcheck's value flow drops that lower
# bound, assumes the copy loop runs zero times, and calls the local
# uninitialised.  The length passed to out() is the same count, so a
# zero-length case would read nothing either way.
uninitvar:src/stdio/printf.c

# SIG_DFL/SIG_IGN/SIG_ERR are (void (*)(int))0/1/-1 by definition -- the
# integer-literal-to-pointer casts C requires of <signal.h>, not a
# portability hazard we can write our way out of.
intToPointerCast:src/signal/signal.c
# system() passes SIG_IGN to signal() while it runs the child, same
# integer-literal-to-pointer cast as above.
intToPointerCast:src/stdlib/system.c

# CLIENT_ID.UniqueProcess/UniqueThread are declared HANDLE by NT but hold
# small integer ids, which is what a pid_t is; the cast back is the
# documented way to read them.  Nothing is truncated: pid_t and the id
# space are both 32-bit on either win32 target.
CastAddressToIntegerAtReturn:src/unistd/getpid.c

# memmove's overlap test is musl's: the unsigned wrap in `- 2*n` is the
# point of it, computing the "distance is at least n in either direction"
# predicate in one comparison.
signConversion:src/string/memmove.c

# `(int)hi >> 31` in __ashrdi3 is an arithmetic right shift of a 32-bit
# signed value by 31, which is implementation-defined (C99 6.5.7p5), not
# undefined -- only a count >= the width would be.  cppcheck applies its
# left-shift rule to `>>` as well and misreports it; verified against
# cppcheck 2.21.1 with a two-line test case.
shiftTooManyBitsSigned:arch/*/src/int64.c

# cppcheck's bundled posix.cfg models fetestexcept()/feclearexcept()/
# feraiseexcept() after glibc's <fenv.h>: a 6-bit FE_ALL_EXCEPT (glibc's
# __FE_DENORM is bit 1) and an argument range annotation that does not
# match FE_ALL_EXCEPT==0x3d (61) at all ("valid values are '1:31'" --
# stale/wrong even for glibc, which allows 0x3f).  include/fenv.h is a
# different, C99-conformant implementation for this project's own
# arches (see its header comment): a 5-bit FE_ALL_EXCEPT with no
# FE_DENORM, since FE_DENORM is not part of the C99 exception set this
# project claims. Calling any of these three with FE_ALL_EXCEPT is
# exactly correct for *our* <fenv.h>, not a bug. The same stale range
# annotation also misfires on a lone FE_INEXACT (0x20 == 32, still
# outside the bogus '1:31' the annotation allows) -- nearbyint.c's
# snapshot-then-clear-just-FE_INEXACT pattern is the one other caller
# in this tree that passes a single-bit argument outside FE_ALL_EXCEPT.
invalidFunctionArg:src/math/fenv.c
invalidFunctionArg:src/math/nearbyint.c
